import "../lib/monaco";
import { useEffect, useMemo, useState } from "react";
import MonacoEditor from "@monaco-editor/react";

type EditorLanguage =
  | "javascript"
  | "jsx"
  | "css"
  | "html"
  | "typescript"
  | "json"
  | "markdown"
  | "plaintext";

type FileNode = {
  id: string;
  type: "file";
  name: string;
  content: string;
  language: EditorLanguage;
  createdAt: number;
};

type FolderNode = {
  id: string;
  type: "folder";
  name: string;
  createdAt: number;
  children: TreeNode[];
};

type TreeNode = FileNode | FolderNode;

type ListedFile = {
  id: string;
  name: string;
  language: EditorLanguage;
  content: string;
  createdAt: number;
  path: string;
};

const LANGUAGE_OPTIONS: { value: EditorLanguage; label: string }[] = [
  { value: "javascript", label: "JavaScript" },
  { value: "jsx", label: "JSX" },
  { value: "css", label: "CSS" },
  { value: "html", label: "HTML" },
];

const detectLanguage = (fileName: string): EditorLanguage => {
  const lower = fileName.toLowerCase();
  if (lower.endsWith(".tsx")) return "typescript";
  if (lower.endsWith(".jsx")) return "jsx";
  if (lower.endsWith(".ts")) return "typescript";
  if (lower.endsWith(".js")) return "javascript";
  if (lower.endsWith(".json")) return "json";
  if (lower.endsWith(".css")) return "css";
  if (lower.endsWith(".html") || lower.endsWith(".htm")) return "html";
  if (lower.endsWith(".md") || lower.endsWith(".markdown")) return "markdown";
  return "javascript";
};

const makeId = (name: string) =>
  `${name}-${Date.now()}-${Math.random().toString(16).slice(2)}`;

const createStarterTree = (): FolderNode => ({
  id: "root",
  name: "root",
  type: "folder",
  createdAt: Date.now(),
  children: [
    {
      id: "app-js",
      type: "file",
      name: "app.js",
      content: "console.log('Hello, world!');",
      language: "javascript",
      createdAt: Date.now() - 2000,
    },
    {
      id: "index-html",
      type: "file",
      name: "index.html",
      content: "<p>Hello</p>",
      language: "html",
      createdAt: Date.now() - 1000,
    },
  ],
});

const STARTER_TREE = createStarterTree();

const collectFiles = (
  node: TreeNode,
  parentPath = ""
): ListedFile[] => {
  const isRoot = node.type === "folder" && node.id === "root";
  const currentPath = isRoot
    ? parentPath
    : parentPath
    ? `${parentPath}/${node.name}`
    : node.name;

  if (node.type === "file") {
    return [
      {
        id: node.id,
        name: node.name,
        language: node.language,
        content: node.content,
        createdAt: node.createdAt,
        path: currentPath || node.name,
      },
    ];
  }

  return node.children.flatMap((child) =>
    collectFiles(child, currentPath)
  );
};

const findFolderById = (node: TreeNode, targetId: string): FolderNode | null => {
  if (node.type === "folder" && node.id === targetId) return node;
  if (node.type === "folder") {
    for (const child of node.children) {
      const found = findFolderById(child, targetId);
      if (found) return found;
    }
  }
  return null;
};

const findNodeWithParent = (
  node: TreeNode,
  targetId: string,
  parent: FolderNode | null = null
): { node: TreeNode; parent: FolderNode | null } | null => {
  if (node.id === targetId) return { node, parent };
  if (node.type === "folder") {
    for (const child of node.children) {
      const found = findNodeWithParent(child, targetId, node);
      if (found) return found;
    }
  }
  return null;
};

const updateNodeInTree = (
  node: TreeNode,
  targetId: string,
  updater: (node: TreeNode) => TreeNode
): { node: TreeNode; updated: boolean } => {
  if (node.id === targetId) {
    return { node: updater(node), updated: true };
  }

  if (node.type === "folder") {
    let changed = false;
    const children = node.children.map((child) => {
      const result = updateNodeInTree(child, targetId, updater);
      if (result.updated) changed = true;
      return result.node;
    });

    if (changed) {
      return { node: { ...node, children }, updated: true };
    }
  }

  return { node, updated: false };
};

const addChildToTree = (
  node: TreeNode,
  parentId: string,
  child: TreeNode
): TreeNode => {
  if (node.type === "folder" && node.id === parentId) {
    return {
      ...node,
      children: [...node.children, child],
    };
  }

  if (node.type === "folder") {
    return {
      ...node,
      children: node.children.map((c) => addChildToTree(c, parentId, child)),
    };
  }

  return node;
};

const removeNodeFromTree = (
  node: TreeNode,
  targetId: string
): { node: TreeNode; removed: boolean } => {
  if (node.type === "folder") {
    const keptChildren: TreeNode[] = [];
    let removed = false;

    for (const child of node.children) {
      if (child.id === targetId) {
        removed = true;
        continue;
      }
      const result = removeNodeFromTree(child, targetId);
      if (result.removed) removed = true;
      keptChildren.push(result.node);
    }

    return {
      node: { ...node, children: keptChildren },
      removed,
    };
  }

  return { node, removed: false };
};

const filterTree = (node: TreeNode, query: string): TreeNode | null => {
  const match = node.name.toLowerCase().includes(query.toLowerCase());

  if (node.type === "file") {
    return match ? node : null;
  }

  const filteredChildren = node.children
    .map((child) => filterTree(child, query))
    .filter(Boolean) as TreeNode[];

  if (match || filteredChildren.length > 0) {
    return { ...node, children: filteredChildren };
  }

  return null;
};

const sortChildren = (
  children: TreeNode[],
  sortBy: "alpha" | "recent"
): TreeNode[] => {
  const sorted = [...children].sort((a, b) => {
    if (a.type !== b.type) return a.type === "folder" ? -1 : 1;
    if (sortBy === "recent") return b.createdAt - a.createdAt;
    return a.name.localeCompare(b.name);
  });
  return sorted;
};

const getPathSegments = (
  node: TreeNode,
  targetId: string,
  trail: string[] = []
): string[] | null => {
  const nextTrail =
    node.type === "folder" && node.id === "root"
      ? trail
      : [...trail, node.name];

  if (node.id === targetId) return nextTrail;

  if (node.type === "folder") {
    for (const child of node.children) {
      const found = getPathSegments(child, targetId, nextTrail);
      if (found) return found;
    }
  }
  return null;
};

const getPathLabel = (root: FolderNode, targetId: string) => {
  const segments = getPathSegments(root, targetId);
  if (!segments) return null;
  if (segments.length === 0) return "root";
  return segments.join("/");
};

const uniqueNameInFolder = (siblings: TreeNode[], baseName: string) => {
  const match = baseName.match(/^(.*?)(\.[^.]*)?$/);
  const stem = match?.[1] ?? baseName;
  const ext = match?.[2] ?? "";

  let index = 1;
  let candidate = baseName;

  const nameExists = (name: string) =>
    siblings.some((node) => node.name === name);

  while (nameExists(candidate)) {
    candidate = `${stem} (${index})${ext}`;
    index += 1;
  }

  return candidate;
};

export default function EditorPage() {
  const [tree, setTree] = useState<FolderNode>(STARTER_TREE);
  const [expandedIds, setExpandedIds] = useState<Set<string>>(new Set(["root"]));
  const [activeFileId, setActiveFileId] = useState<string | null>(() => {
    const first = collectFiles(STARTER_TREE)[0];
    return first ? first.id : null;
  });
  const [creationTargetId, setCreationTargetId] = useState<string>("root");
  const [filter, setFilter] = useState("");
  const [sortBy, setSortBy] = useState<"alpha" | "recent">("alpha");
  const [renamingId, setRenamingId] = useState<string | null>(null);
  const [renameValue, setRenameValue] = useState("");
  const [openMenuId, setOpenMenuId] = useState<string | null>(null);
  const [newFileName, setNewFileName] = useState("");
  const [newFileContent, setNewFileContent] = useState("");
  const [newFileLanguage, setNewFileLanguage] =
    useState<EditorLanguage>("javascript");
  const [languageManuallyChosen, setLanguageManuallyChosen] = useState(false);
  const [newFolderName, setNewFolderName] = useState("");

  const allFiles = useMemo(() => collectFiles(tree), [tree]);
  const activeFile = allFiles.find((file) => file.id === activeFileId) || null;

  useEffect(() => {
    if (activeFileId && !activeFile) {
      setActiveFileId(allFiles[0]?.id ?? null);
    }
  }, [activeFileId, activeFile, allFiles]);

  const visibleTree = useMemo(() => {
    if (!filter.trim()) return tree;
    const filtered = filterTree(tree, filter.trim());
    if (filtered && filtered.type === "folder") return filtered;
    return { ...tree, children: [] };
  }, [tree, filter]);

  const targetFolder =
    findFolderById(tree, creationTargetId) || findFolderById(tree, "root")!;
  const targetFolderLabel =
    getPathLabel(tree, creationTargetId) || "root";

  const activeMonacoLanguage =
    activeFile?.language === "jsx" ? "javascript" : activeFile?.language;
  const needsJsxPathExtension =
    activeFile?.language === "jsx" &&
    activeFile?.name &&
    !/\.(jsx|tsx)$/i.test(activeFile.name);
  const activePath = needsJsxPathExtension
    ? `${activeFile?.path}.jsx`
    : activeFile?.path;

  const toggleExpanded = (id: string) => {
    setExpandedIds((prev) => {
      const next = new Set(prev);
      if (next.has(id)) next.delete(id);
      else next.add(id);
      return next;
    });
  };

  const setCreationTarget = (id: string) => {
    const folder = findFolderById(tree, id);
    if (!folder) return;
    setCreationTargetId(id);
    setExpandedIds((prev) => {
      const next = new Set(prev);
      next.add(id);
      return next;
    });
  };

  const handleCreateFolder = () => {
    const trimmed = newFolderName.trim();
    if (!trimmed) {
      alert("Folder name cannot be empty.");
      return;
    }

    const parentFolder = targetFolder;
    if (parentFolder.children.some((child) => child.name === trimmed)) {
      alert("A folder or file with that name already exists in this folder.");
      return;
    }

    const folder: FolderNode = {
      id: makeId(trimmed),
      type: "folder",
      name: trimmed,
      createdAt: Date.now(),
      children: [],
    };

    setTree((prev) => addChildToTree(prev, parentFolder.id, folder) as FolderNode);
    setExpandedIds((prev) => {
      const next = new Set(prev);
      next.add(parentFolder.id);
      next.add(folder.id);
      return next;
    });
    setCreationTargetId(folder.id);
    setNewFolderName("");
  };

  const handleCreateFile = () => {
    const trimmedName = newFileName.trim();
    if (!trimmedName) {
      alert("File name cannot be empty.");
      return;
    }

    const parentFolder = targetFolder;
    if (parentFolder.children.some((child) => child.name === trimmedName)) {
      alert("A file or folder with that name already exists in this folder.");
      return;
    }

    const language =
      languageManuallyChosen || !trimmedName
        ? newFileLanguage
        : detectLanguage(trimmedName);

    const file: FileNode = {
      id: makeId(trimmedName),
      type: "file",
      name: trimmedName,
      content: newFileContent,
      language,
      createdAt: Date.now(),
    };

    setTree((prev) => addChildToTree(prev, parentFolder.id, file) as FolderNode);
    setActiveFileId(file.id);
    setExpandedIds((prev) => {
      const next = new Set(prev);
      next.add(parentFolder.id);
      return next;
    });
    setNewFileName("");
    setNewFileContent("");
    setLanguageManuallyChosen(false);
    setNewFileLanguage("javascript");
  };

  const handleEditorChange = (value?: string) => {
    if (!activeFileId) return;

    const updated = updateNodeInTree(tree, activeFileId, (node) => {
      if (node.type === "file") {
        return { ...node, content: value ?? "" };
      }
      return node;
    });

    if (updated.updated) setTree(updated.node as FolderNode);
  };

  const handleLanguageChange = (language: EditorLanguage) => {
    if (!activeFileId) return;

    const updated = updateNodeInTree(tree, activeFileId, (node) => {
      if (node.type === "file") {
        return { ...node, language };
      }
      return node;
    });

    if (updated.updated) setTree(updated.node as FolderNode);
  };

  const handleDuplicateFile = (id: string) => {
    const found = findNodeWithParent(tree, id);
    if (!found || found.node.type !== "file") return;

    const parentFolder = found.parent ?? findFolderById(tree, "root")!;
    const siblings = parentFolder.children;
    const copyName = uniqueNameInFolder(siblings, found.node.name);

    const duplicate: FileNode = {
      ...found.node,
      id: makeId(copyName),
      name: copyName,
      createdAt: Date.now(),
    };

    setTree((prev) => addChildToTree(prev, parentFolder.id, duplicate) as FolderNode);
    setActiveFileId(duplicate.id);
    setExpandedIds((prev) => {
      const next = new Set(prev);
      next.add(parentFolder.id);
      return next;
    });
    setOpenMenuId(null);
  };

  const handleDeleteNode = (id: string) => {
    if (id === "root") return;

    const result = removeNodeFromTree(tree, id);
    if (result.removed) {
      setTree(result.node as FolderNode);
      setOpenMenuId(null);
      if (renamingId === id) {
        setRenamingId(null);
        setRenameValue("");
      }
    }
  };

  const handleRenameSave = () => {
    if (!renamingId) return;
    const trimmed = renameValue.trim();
    if (!trimmed) {
      alert("Name cannot be empty.");
      return;
    }

    const info = findNodeWithParent(tree, renamingId);
    if (!info || info.node.id === "root") {
      setRenamingId(null);
      setRenameValue("");
      return;
    }

    const parentFolder = info.parent ?? findFolderById(tree, "root")!;
    const siblings = parentFolder.children;
    const duplicate = siblings.some(
      (node) => node.name === trimmed && node.id !== renamingId
    );
    if (duplicate) {
      alert("A sibling with that name already exists.");
      return;
    }

    const updated = updateNodeInTree(tree, renamingId, (node) => {
      if (node.type === "file") {
        return { ...node, name: trimmed, language: detectLanguage(trimmed) };
      }
      return { ...node, name: trimmed };
    });

    if (updated.updated) setTree(updated.node as FolderNode);
    setRenamingId(null);
    setRenameValue("");
    setOpenMenuId(null);
  };

  const renderNode = (
    node: TreeNode,
    depth: number,
    parentPath = ""
  ): JSX.Element => {
    const isFolder = node.type === "folder";
    const isExpanded = isFolder ? expandedIds.has(node.id) : false;
    const isActiveFile = activeFileId === node.id;
    const isRenaming = renamingId === node.id;
    const nodePath = parentPath
      ? `${parentPath}/${node.name}`
      : node.name;
    const children =
      isFolder && node.children
        ? sortChildren(node.children, sortBy)
        : [];

    return (
      <div key={node.id}>
        <div
          className={`group flex items-center gap-2 rounded px-2 py-1 text-sm transition ${
            isActiveFile ? "bg-blue-100 text-blue-800" : "hover:bg-gray-100"
          }`}
          style={{ paddingLeft: depth * 12 }}
        >
          {isFolder ? (
            <button
              onClick={() => {
                toggleExpanded(node.id);
                setCreationTarget(node.id);
              }}
              className="flex items-center gap-2 text-left"
            >
              <span className="text-xs">{isExpanded ? "▾" : "▸"}</span>
              <span className="font-semibold">{node.name}</span>
            </button>
          ) : isRenaming ? null : (
            <button
              onClick={() => setActiveFileId(node.id)}
              className="flex flex-1 items-center gap-2 text-left"
            >
              <span className="text-xs text-gray-500">•</span>
              <span className="truncate">{node.name}</span>
            </button>
          )}

          {isRenaming ? (
            <div className="flex flex-1 items-center gap-2">
              <input
                autoFocus
                value={renameValue}
                onChange={(e) => setRenameValue(e.target.value)}
                onKeyDown={(e) => {
                  if (e.key === "Enter") handleRenameSave();
                  if (e.key === "Escape") {
                    setRenamingId(null);
                    setRenameValue("");
                  }
                }}
                className="flex-1 rounded border border-gray-200 px-2 py-1 text-sm focus:border-blue-400 focus:outline-none"
              />
              <button
                onClick={handleRenameSave}
                className="rounded bg-blue-600 px-2 py-1 text-xs font-semibold text-white hover:bg-blue-700"
              >
                Save
              </button>
              <button
                onClick={() => {
                  setRenamingId(null);
                  setRenameValue("");
                }}
                className="rounded border border-gray-200 px-2 py-1 text-xs font-semibold text-gray-700 hover:bg-gray-100"
              >
                Cancel
              </button>
            </div>
          ) : (
            <div className="ml-auto flex items-center gap-2">
              {isFolder && creationTargetId === node.id && (
                <span className="rounded bg-blue-100 px-2 py-0.5 text-[11px] font-semibold text-blue-700">
                  target
                </span>
              )}
              {node.type === "file" && (
                <span className="rounded bg-gray-100 px-2 py-0.5 text-[11px] font-semibold uppercase text-gray-600">
                  {node.language}
                </span>
              )}
              <div className="relative">
                <button
                  onClick={() =>
                    setOpenMenuId((prev) =>
                      prev === node.id ? null : node.id
                    )
                  }
                  className="rounded px-2 py-1 text-xs text-gray-500 transition hover:bg-gray-200"
                >
                  ⋯
                </button>
                {openMenuId === node.id && (
                  <div className="absolute right-0 z-10 mt-1 w-44 overflow-hidden rounded border border-gray-200 bg-white shadow">
                    {isFolder && (
                      <>
                        <button
                          onClick={() => {
                            setCreationTarget(node.id);
                            setOpenMenuId(null);
                          }}
                          className="block w-full px-3 py-2 text-left text-sm hover:bg-gray-50"
                        >
                          Use for new items
                        </button>
                        {node.id !== "root" && (
                          <button
                            onClick={() => {
                              setRenamingId(node.id);
                              setRenameValue(node.name);
                              setOpenMenuId(null);
                            }}
                            className="block w-full px-3 py-2 text-left text-sm hover:bg-gray-50"
                          >
                            Rename folder
                          </button>
                        )}
                        {node.id !== "root" && (
                          <button
                            onClick={() => handleDeleteNode(node.id)}
                            className="block w-full px-3 py-2 text-left text-sm text-red-600 hover:bg-red-50"
                          >
                            Delete folder
                          </button>
                        )}
                      </>
                    )}
                    {node.type === "file" && (
                      <>
                        <button
                          onClick={() => {
                            setActiveFileId(node.id);
                            setOpenMenuId(null);
                          }}
                          className="block w-full px-3 py-2 text-left text-sm hover:bg-gray-50"
                        >
                          Open
                        </button>
                        <button
                          onClick={() => {
                            setRenamingId(node.id);
                            setRenameValue(node.name);
                            setOpenMenuId(null);
                          }}
                          className="block w-full px-3 py-2 text-left text-sm hover:bg-gray-50"
                        >
                          Rename file
                        </button>
                        <button
                          onClick={() => handleDuplicateFile(node.id)}
                          className="block w-full px-3 py-2 text-left text-sm hover:bg-gray-50"
                        >
                          Duplicate
                        </button>
                        <button
                          onClick={() => handleDeleteNode(node.id)}
                          className="block w-full px-3 py-2 text-left text-sm text-red-600 hover:bg-red-50"
                        >
                          Delete
                        </button>
                      </>
                    )}
                  </div>
                )}
              </div>
            </div>
          )}
        </div>

        {isFolder && isExpanded && (
          <div className="flex flex-col">
            {children.map((child) => renderNode(child, depth + 1, nodePath))}
          </div>
        )}
      </div>
    );
  };

  return (
    <div className="flex h-screen bg-gray-50 text-gray-900">
      <aside className="w-80 border-r border-gray-200 bg-white p-4">
        <div className="mb-3 flex items-center justify-between">
          <h2 className="text-lg font-semibold">Explorer</h2>
          <span className="rounded bg-gray-100 px-2 py-0.5 text-[11px] font-semibold text-gray-600">
            {sortBy === "alpha" ? "A-Z" : "Recent"}
          </span>
        </div>
        <div className="flex flex-col gap-3">
          <input
            value={filter}
            onChange={(e) => setFilter(e.target.value)}
            placeholder="Search files..."
            className="w-full rounded border border-gray-200 px-2 py-1 text-sm focus:border-blue-400 focus:outline-none"
          />
          <div className="flex items-center gap-2 text-xs text-gray-600">
            <span>Sort</span>
            <select
              value={sortBy}
              onChange={(e) => setSortBy(e.target.value as "alpha" | "recent")}
              className="rounded border border-gray-200 px-2 py-1 text-sm focus:border-blue-400 focus:outline-none"
            >
              <option value="alpha">A-Z</option>
              <option value="recent">Recent first</option>
            </select>
          </div>
          <div className="rounded border border-gray-200 bg-gray-50 p-2">
            <div className="mb-1 flex items-center justify-between text-xs text-gray-600">
              <span>FILES</span>
              <span className="text-gray-400">
                Target: {targetFolderLabel}
              </span>
            </div>
            <div className="max-h-[50vh] overflow-auto pr-1">
              {renderNode(visibleTree, 0)}
            </div>
          </div>

          <div className="space-y-2 rounded-md border border-gray-200 bg-gray-50 p-3">
            <h3 className="text-sm font-semibold">New folder</h3>
            <label className="block text-xs font-medium text-gray-600">
              Folder name
            </label>
            <input
              value={newFolderName}
              onChange={(e) => setNewFolderName(e.target.value)}
              placeholder="my-folder"
              className="w-full rounded border border-gray-200 px-2 py-1 text-sm focus:border-blue-400 focus:outline-none"
            />
            <button
              onClick={handleCreateFolder}
              className="w-full rounded bg-gray-800 px-3 py-2 text-sm font-semibold text-white transition hover:bg-gray-900"
            >
              Add folder
            </button>
          </div>

          <div className="space-y-2 rounded-md border border-gray-200 bg-gray-50 p-3">
            <h3 className="text-sm font-semibold">New file</h3>
            <label className="block text-xs font-medium text-gray-600">
              File name
            </label>
            <input
              value={newFileName}
              onChange={(e) => {
                setNewFileName(e.target.value);
                if (!languageManuallyChosen) {
                  setNewFileLanguage(detectLanguage(e.target.value));
                }
              }}
              placeholder="example.ts"
              className="w-full rounded border border-gray-200 px-2 py-1 text-sm focus:border-blue-400 focus:outline-none"
            />
            <label className="block text-xs font-medium text-gray-600">
              Language
            </label>
            <select
              value={newFileLanguage}
              onChange={(e) => {
                setLanguageManuallyChosen(true);
                setNewFileLanguage(e.target.value as EditorLanguage);
              }}
              className="w-full rounded border border-gray-200 px-2 py-1 text-sm focus:border-blue-400 focus:outline-none"
            >
              {LANGUAGE_OPTIONS.map((option) => (
                <option key={option.value} value={option.value}>
                  {option.label}
                </option>
              ))}
            </select>
            <label className="block text-xs font-medium text-gray-600">
              Starting content
            </label>
            <textarea
              value={newFileContent}
              onChange={(e) => setNewFileContent(e.target.value)}
              className="h-24 w-full resize-none rounded border border-gray-200 px-2 py-1 text-sm focus:border-blue-400 focus:outline-none"
              placeholder="// Optional starter code"
            />
            <button
              onClick={handleCreateFile}
              className="w-full rounded bg-blue-600 px-3 py-2 text-sm font-semibold text-white transition hover:bg-blue-700"
            >
              Add file
            </button>
          </div>
        </div>
      </aside>

      <main className="flex flex-1 flex-col">
        <div className="flex items-center gap-3 border-b border-gray-200 bg-white px-4 py-3">
          <div className="flex flex-wrap items-center gap-2">
            {allFiles.map((file) => (
              <button
                key={file.id}
                onClick={() => setActiveFileId(file.id)}
                className={`rounded px-3 py-1 text-sm transition hover:bg-blue-50 ${
                  activeFileId === file.id
                    ? "bg-blue-100 font-semibold text-blue-700"
                    : "text-gray-700"
                }`}
              >
                {file.path}
              </button>
            ))}
          </div>

          {activeFile && (
            <div className="ml-auto flex items-center gap-2 text-xs text-gray-600">
              <span>Language</span>
              <select
                value={
                  LANGUAGE_OPTIONS.some(
                    (option) => option.value === activeFile.language
                  )
                    ? activeFile.language
                    : "javascript"
                }
                onChange={(e) =>
                  handleLanguageChange(e.target.value as EditorLanguage)
                }
                className="rounded border border-gray-200 px-2 py-1 text-sm text-gray-700 focus:border-blue-400 focus:outline-none"
              >
                {LANGUAGE_OPTIONS.map((option) => (
                  <option key={option.value} value={option.value}>
                    {option.label}
                  </option>
                ))}
              </select>
            </div>
          )}
        </div>

        <div className="flex flex-1 flex-col bg-[#1e1e1e] min-h-0">
          {activeFile ? (
            <MonacoEditor
              height="100%"
              path={activePath}
              language={activeMonacoLanguage}
              theme="vs-dark"
              value={activeFile.content}
              onChange={handleEditorChange}
              options={{
                fontSize: 14,
                minimap: { enabled: false },
                scrollBeyondLastLine: false,
                automaticLayout: true,
              }}
            />
          ) : (
            <div className="flex flex-1 items-center justify-center text-sm text-gray-500">
              Select or create a file to start editing.
            </div>
          )}
        </div>
      </main>
    </div>
  );
}
